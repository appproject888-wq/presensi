/** * PRESENSI WAJAH - backend (Google Apps Script) * * Dipasang di dalam Google Sheets (Extensions -> Apps Script). * Sheet yang dibutuhkan: * Pegawai : ID | Nama | Divisi | Wajah | Aktif * Presensi : Tanggal | ID | Nama | Masuk | Pulang | Kecocokan | Ms_Terakhir * * Script Properties (Project Settings -> Script Properties): * API_TOKEN : token rahasia, harus sama dengan di halaman presensi * ADMIN_PIN : PIN untuk mode daftar (4-6 angka) */ const SHEET_PEGAWAI = 'Pegawai'; const SHEET_PRESENSI = 'Presensi'; const MATCH_THRESHOLD = 0.5; // jarak maksimum dianggap cocok const MARGIN = 0.05; // selisih minimum antara pegawai terbaik dan kedua const COOLDOWN_MENIT = 5; // jeda minimum sebelum scan dicatat ulang const MAKS_WAJAH_PER_KIRIM = 10; // batas wajah dalam satu kiriman const TZ = 'Asia/Jakarta'; function doGet() { return json_({ ok: true, info: 'API presensi aktif' }); } function doPost(e) { try { const body = JSON.parse(e.postData.contents); const props = PropertiesService.getScriptProperties(); if (body.token !== props.getProperty('API_TOKEN')) { return json_({ ok: false, error: 'Token salah' }); } if (body.action === 'attend') return json_(absenBanyak_(body.vecs)); if (body.action === 'enroll') return json_(daftar_(body)); return json_({ ok: false, error: 'Aksi tidak dikenal' }); } catch (err) { return json_({ ok: false, error: String(err) }); } } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function validVec_(vec) { return Array.isArray(vec) && vec.length === 128 && vec.every(function (v) { return typeof v === 'number' && isFinite(v); }); } // jarak Euclidean antara dua descriptor wajah function jarak_(a, b) { let s = 0; for (let i = 0; i < a.length; i++) { const d = a[i] - b[i]; s += d * d; } return Math.sqrt(s); } // ubah isi sel jadi teks (Sheets kadang mengubah tanggal/angka otomatis) function teks_(v) { return v instanceof Date ? Utilities.formatDate(v, TZ, 'yyyy-MM-dd') : String(v).trim(); } function loadPegawai_() { const rows = SpreadsheetApp.getActive().getSheetByName(SHEET_PEGAWAI).getDataRange().getValues(); const hasil = []; for (let i = 1; i < rows.length; i++) { const id = teks_(rows[i][0]); const nama = teks_(rows[i][1]); const wajah = rows[i][3]; const aktif = teks_(rows[i][4]).toLowerCase(); if (!id || !wajah || aktif === 'tidak') continue; try { hasil.push({ id: id, nama: nama, vec: JSON.parse(wajah) }); } catch (_) { // baris dengan data wajah rusak dilewati } } return hasil; } // cari pegawai yang cocok untuk satu wajah; null berarti belum terdaftar function cariPegawai_(vec, pegawai) { const urut = pegawai .map(function (p) { return { p: p, d: jarak_(vec, p.vec) }; }) .sort(function (a, b) { return a.d - b.d; }); const terbaik = urut[0]; const kedua = urut[1]; if (terbaik.d > MATCH_THRESHOLD) return null; if (kedua && (kedua.d - terbaik.d) < MARGIN) return null; const kecocokan = Math.round(Math.max(0, 1 - terbaik.d) * 1000) / 10; // contoh: 92.3 return { p: terbaik.p, kecocokan: kecocokan }; } // proses banyak wajah sekaligus dalam satu kiriman function absenBanyak_(vecs) { if (!Array.isArray(vecs)) return { ok: false, error: 'Data wajah tidak valid' }; vecs = vecs.slice(0, MAKS_WAJAH_PER_KIRIM); const pegawai = loadPegawai_(); if (pegawai.length === 0) return { ok: true, hasil: [{ status: 'kosong' }] }; const sh = SpreadsheetApp.getActive().getSheetByName(SHEET_PRESENSI); const rows = sh.getDataRange().getValues(); const now = new Date(); const tanggal = Utilities.formatDate(now, TZ, 'yyyy-MM-dd'); const jam = Utilities.formatDate(now, TZ, 'HH:mm:ss'); const nowMs = now.getTime(); const sudahDiproses = {}; const hasil = []; vecs.forEach(function (vec) { if (!validVec_(vec)) { hasil.push({ status: 'error', error: 'Data wajah tidak valid' }); return; } const c = cariPegawai_(vec, pegawai); if (!c) { hasil.push({ status: 'tidak_dikenali' }); return; } const p = c.p; if (sudahDiproses[p.id]) return; // orang yang sama dua kali dalam satu kiriman sudahDiproses[p.id] = true; // cari baris presensi pegawai ini hari ini let baris = -1; for (let i = 1; i < rows.length; i++) { if (teks_(rows[i][0]) === tanggal && teks_(rows[i][1]) === p.id) { baris = i; break; } } if (baris !== -1) { const terakhir = Number(rows[baris][6]) || 0; if (nowMs - terakhir < COOLDOWN_MENIT * 60000) { hasil.push({ status: 'cooldown', id: p.id, nama: p.nama, kecocokan: c.kecocokan }); return; } sh.getRange(baris + 1, 5).setValue(jam); // kolom Pulang sh.getRange(baris + 1, 6).setValue(c.kecocokan); // kolom Kecocokan sh.getRange(baris + 1, 7).setValue(nowMs); // kolom Ms_Terakhir hasil.push({ status: 'pulang', id: p.id, nama: p.nama, jam: jam, kecocokan: c.kecocokan }); return; } sh.appendRow([tanggal, p.id, p.nama, jam, '', c.kecocokan, nowMs]); hasil.push({ status: 'masuk', id: p.id, nama: p.nama, jam: jam, kecocokan: c.kecocokan }); }); return { ok: true, hasil: hasil }; } function daftar_(body) { const pin = PropertiesService.getScriptProperties().getProperty('ADMIN_PIN'); if (String(body.pin) !== pin) return { ok: false, error: 'PIN admin salah' }; if (!validVec_(body.vec)) return { ok: false, error: 'Data wajah tidak valid' }; const id = String(body.id || '').trim(); const nama = String(body.nama || '').trim(); if (!id || !nama) return { ok: false, error: 'ID dan nama wajib diisi' }; const sh = SpreadsheetApp.getActive().getSheetByName(SHEET_PEGAWAI); const rows = sh.getDataRange().getValues(); const wajah = JSON.stringify(body.vec.map(function (v) { return Math.round(v * 1e6) / 1e6; })); for (let i = 1; i < rows.length; i++) { if (teks_(rows[i][0]) === id) { sh.getRange(i + 1, 2).setValue(nama); sh.getRange(i + 1, 4).setValue(wajah); sh.getRange(i + 1, 5).setValue('Ya'); return { ok: true, status: 'diperbarui', nama: nama }; } } sh.appendRow([id, nama, String(body.divisi || ''), wajah, 'Ya']); return { ok: true, status: 'baru', nama: nama }; }
